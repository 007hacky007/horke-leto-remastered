cmake_minimum_required(VERSION 3.25)
project(horke_leto VERSION 0.0.1 LANGUAGES C CXX)
if(APPLE)
 enable_language(OBJCXX)
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_compile_options("-ffile-prefix-map=${CMAKE_SOURCE_DIR}=." "-fdebug-prefix-map=${CMAKE_SOURCE_DIR}=.")
include(cmake/Dependencies.cmake)
add_library(hl-core src/platform/app_options.cpp src/debug/trace.cpp src/debug/replay.cpp src/debug/checkpoint.cpp)
target_include_directories(hl-core PUBLIC src)
target_link_libraries(hl-core PUBLIC hl-json)
if(WIN32)
    target_compile_definitions(hl-core PUBLIC _WIN32_WINNT=0x0A00 WINVER=0x0A00)
endif()
add_executable(hl src/main.cpp src/platform/app.cpp)
target_include_directories(hl PRIVATE tools/vendor/stb)
target_link_libraries(hl PRIVATE hl-core SDL3::SDL3)
if(WIN32)
    add_custom_command(TARGET hl POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:SDL3::SDL3> $<TARGET_FILE_DIR:hl>)
endif()
add_library(hl-le tools/le/le_image.cpp tools/le/le_fixups.cpp tools/le/le_to_elf.cpp)
target_include_directories(hl-le PUBLIC tools)
target_link_libraries(hl-le PUBLIC hl-core)
add_executable(hl-tool tools/main.cpp)
target_link_libraries(hl-tool PRIVATE hl-le)
include(CTest)
if(BUILD_TESTING)
    add_subdirectory(tests)
endif()
target_sources(hl-core PRIVATE src/data/byte_reader.cpp src/data/disc.cpp)
target_sources(hl-core PRIVATE src/data/hash.cpp)
add_library(hl-assets tools/import.cpp)
target_include_directories(hl-assets PUBLIC tools)
target_link_libraries(hl-assets PUBLIC hl-core)
target_link_libraries(hl-tool PRIVATE hl-assets)
if(BUILD_TESTING)
 target_link_libraries(hl-tests PRIVATE hl-assets)
endif()
target_sources(hl-core PRIVATE src/data/grp.cpp)
target_sources(hl-assets PRIVATE tools/assets.cpp)
target_sources(hl-core PRIVATE src/data/adb.cpp)
target_sources(hl-core PRIVATE src/data/flc.cpp)
target_sources(hl-assets PRIVATE tools/image_io.cpp tools/frames.cpp)
target_include_directories(hl-assets PRIVATE tools/vendor/stb)
# Keep offline hashing and PNG compression practical in debug corpus sweeps.
set_source_files_properties(src/data/hash.cpp tools/image_io.cpp PROPERTIES COMPILE_OPTIONS "-O2")
target_sources(hl-core PRIVATE src/data/wav.cpp)
target_sources(hl-core PRIVATE src/data/text.cpp src/data/font.cpp)
target_sources(hl-core PRIVATE src/data/flac.cpp)
target_include_directories(hl-core PRIVATE tools/vendor/dr_libs)
target_sources(hl-assets PRIVATE tools/cd.cpp)
target_link_libraries(hl-assets PRIVATE SDL3::SDL3)
if(WIN32)
 target_link_libraries(hl-assets PRIVATE ktmw32)
endif()
target_sources(hl-assets PRIVATE tools/font_export.cpp)
target_sources(hl-assets PRIVATE tools/viewer_model.cpp tools/coverage.cpp)
target_sources(hl-assets PRIVATE tools/viewer.cpp)
target_sources(hl-core PRIVATE src/script/opcode.cpp src/script/object.cpp)
target_sources(hl-core PRIVATE src/script/interpreter.cpp)
target_sources(hl-core PRIVATE src/script/scheduler.cpp)
target_sources(hl-assets PRIVATE tools/script_trace.cpp)
target_sources(hl-core PRIVATE src/data/character.cpp)
target_sources(hl-core PRIVATE src/render/scene_graph.cpp src/render/palette.cpp)
target_sources(hl-core PRIVATE src/audio/mixer.cpp src/audio/cd_player.cpp)
target_sources(hl-core PRIVATE src/script/intro_runtime.cpp)
target_sources(hl-core PRIVATE src/script/intro_movement.cpp)
target_sources(hl-core PRIVATE src/render/text_overlay.cpp)
target_sources(hl-core PRIVATE src/data/frame_crop.cpp)
add_library(hl-render src/render/gpu.cpp)
target_link_libraries(hl-render PUBLIC hl-core SDL3::SDL3)
target_link_libraries(hl PRIVATE hl-render)
target_link_libraries(hl-assets PUBLIC hl-render)
target_sources(hl PRIVATE src/platform/native_app.cpp)
target_sources(hl-assets PRIVATE tools/replay.cpp)
target_sources(hl-core PRIVATE src/world/movement.cpp)
target_sources(hl-core PRIVATE src/save/save.cpp)
# Keep original-frame conversion fast enough for the 35 Hz playable-scene gate.
set_source_files_properties(src/script/intro_runtime.cpp src/render/scene_graph.cpp src/data/frame_crop.cpp PROPERTIES COMPILE_OPTIONS "-O2")
target_sources(hl-core PRIVATE src/render/pack.cpp)
target_include_directories(hl-core PRIVATE tools/vendor/stb)
set_source_files_properties(src/render/pack.cpp PROPERTIES COMPILE_OPTIONS "-O2")
target_sources(hl-assets PRIVATE tools/pack.cpp)
target_sources(hl-core PRIVATE src/render/graphics_mode.cpp src/render/active_areas.cpp)
target_sources(hl-assets PRIVATE tools/benchmark.cpp)
if(APPLE)
 target_sources(hl-assets PRIVATE tools/metal_timing.mm)
 set_source_files_properties(tools/metal_timing.mm PROPERTIES COMPILE_OPTIONS "-fobjc-arc")
 target_link_libraries(hl-assets PRIVATE "-framework Metal" "-framework Foundation")
else()
 target_sources(hl-assets PRIVATE tools/gpu_timing.cpp)
endif()
add_subdirectory(packaging)
add_subdirectory(builder)
